# -*- mode: ruby -*-
# vi: set ft=ruby :

Vagrant.configure("2") do |config|

  # Base box
  config.vm.box = "ubuntu/jammy64"

  # Réseau privé (host-only network)
  # Permet la communication entre l’hôte et la VM uniquement.
  # L’IP est fixée manuellement.
  config.vm.network "private_network", ip: "10.2.33.10"

  # Réseau public (bridged network)
  # La machine virtuelle apparaît comme un appareil
  # à part entière sur le réseau local.
  config.vm.network "public_network", bridge: "Intel(R) Ethernet Connection (16) I219-LM"

  # Synced folder: ./scripts sur l'hôte -> /home/vagrant/scripts dans la VM
  config.vm.synced_folder "./scripts", "/opt/scripts"

  # Provisioning: exécuter le script depuis la VM
  # config.vm.provision "shell", path: "/home/vagrant/scripts/bootstrap.sh"

  # VirtualBox provider configuration
  config.vm.provider "virtualbox" do |vb|
    vb.memory = 1024
    vb.cpus  = 2
  end

end
